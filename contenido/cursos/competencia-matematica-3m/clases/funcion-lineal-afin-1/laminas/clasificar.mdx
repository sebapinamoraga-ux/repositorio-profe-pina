<Paneles>
<Tarjeta titulo="f(x) = 4x">
<Paso n={1}>**Lineal:** $m=4$, $n=0$.</Paso>
</Tarjeta>
<Tarjeta titulo="g(x) = 2x − 3">
<Paso n={2}>**Afín:** $m=2$, $n=-3$.</Paso>
</Tarjeta>
</Paneles>
<Paneles>
<Tarjeta titulo="h(x) = −x/2">
<Paso n={3}>**Lineal:** $m=-\tfrac{1}{2}$, $n=0$.</Paso>
</Tarjeta>
<Tarjeta titulo="p(x) = x² + 1">
<Paso n={4}>**Ninguna:** $x$ está al cuadrado; su gráfico no es una recta.</Paso>
</Tarjeta>
</Paneles>
