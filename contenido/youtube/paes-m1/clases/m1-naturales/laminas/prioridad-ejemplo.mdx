<Composicion plantilla="pedagogica-modelado" nota="Un paso por línea: así se ve qué operación va primero.">
<EjemploResuelto>
$$
18-2\cdot(3+2^2):7
$$

<Paso n={1}>
Potencia dentro del paréntesis: $=18-2\cdot(3+4):7$
</Paso>
<Paso n={2}>
Paréntesis: $=18-2\cdot 7:7$
</Paso>
<Paso n={3}>
Multiplicación y división, de izquierda a derecha: $=18-14:7=18-2$
</Paso>
</EjemploResuelto>

<Paso n={4}>
<Comprobacion titulo="Resultado">
$18-2=16$.
</Comprobacion>
</Paso>
</Composicion>
