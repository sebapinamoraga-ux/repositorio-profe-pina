$$
\begin{array}{c|l}
\text{por} & \text{un número es divisible si}\\ \hline
2 & \text{termina en } 0,\ 2,\ 4,\ 6 \text{ u } 8\\
3 & \text{la suma de sus cifras es múltiplo de } 3\\
5 & \text{termina en } 0 \text{ o } 5\\
9 & \text{la suma de sus cifras es múltiplo de } 9\\
10 & \text{termina en } 0
\end{array}
$$
