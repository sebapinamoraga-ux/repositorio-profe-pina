<Composicion plantilla="pedagogica-modelado" nota="Sumamos miembro a miembro: los coeficientes de x son opuestos.">
<Tarjeta titulo="Ecuaciones preparadas">

$$
\begin{array}{rrrrr}&-4x&-6y&=&-10.200\\+&4x&+y&=&6.700\\\hline\end{array}
$$

</Tarjeta>
<Paso n={1}>

$$
\underbrace{-4x+4x}_{0}+\underbrace{-6y+y}_{-5y}=-10.200+6.700
$$

</Paso>
<Paso n={2}>
<Propiedad titulo="Una ecuación con una incógnita">

$$
-5y=-3.500\quad\Rightarrow\quad y=700
$$

</Propiedad>
</Paso>
</Composicion>
